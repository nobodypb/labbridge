cmake_minimum_required(VERSION 3.20)
project(labbridge_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(labbridge_core firmware/drivers/rd60xx/rd_wifi.cpp firmware/drivers/meters/ina228.cpp firmware/drivers/meters/meter_driver.cpp firmware/drivers/meters/yocto_watt.cpp firmware/core/src/components.cpp firmware/core/src/resources.cpp firmware/drivers/dl24m/codec.cpp firmware/drivers/dl24m/dl24m_driver.cpp firmware/drivers/rd60xx/rd60xx.cpp firmware/drivers/rd60xx/rd60xx_driver.cpp firmware/core/src/node.cpp firmware/core/src/captive_dns.cpp firmware/core/src/stream.cpp)
target_include_directories(labbridge_core PUBLIC firmware/drivers/meters firmware/core/include firmware/drivers/dl24m firmware/drivers/rd60xx firmware/generated)
target_compile_options(labbridge_core PRIVATE -Wall -Wextra -Werror)
file(GLOB API_MODELS CONFIGURE_DEPENDS firmware/generated/c/model/*.c)
add_library(labbridge_api_models ${API_MODELS} firmware/generated/c/src/list.c firmware/generated/c/src/binary.c firmware/generated/c/external/cJSON.c)
target_include_directories(labbridge_api_models PUBLIC firmware/generated/c/model firmware/generated/c/include firmware/generated/c/external)
target_link_libraries(labbridge_core PUBLIC labbridge_api_models)
enable_testing()
foreach(test resources dl24m_codec interlock temperature transaction guard rd60xx node dl24m_driver rd60xx_driver cell_guard interlock_delayed catalog timing driver_queue configuration_safety json output_deadline components meters rd_wifi stream dl24m_extended dl24m_modes stream_rpc json_allocation stream_memory)
 add_executable(test_${test} tests/test_${test}.cpp)
 target_link_libraries(test_${test} PRIVATE labbridge_core)
 add_test(NAME ${test} COMMAND test_${test})
endforeach()
add_executable(contract_dump tests/contract_dump.cpp)
target_link_libraries(contract_dump PRIVATE labbridge_core)

target_compile_definitions(test_rd_wifi PRIVATE LABBRIDGE_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
